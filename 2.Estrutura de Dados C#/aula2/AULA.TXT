
string[] nomes = new string[5];
double[] alturas = new double[5];

int indice = 0;


void Push(string nome, double altura)
{
    if (indice < 5)
    {
        nomes[indice] = nome;
        alturas[indice] = altura;

        Console.WriteLine(nome + "Foi adicionado a pilha");
        indice++;
    }
    else
    {
        Console.WriteLine("A pilha está cheia.");
    }
}

void Pop()
{
    if (indice > 0)
    {
        indice--;

        Console.WriteLine(nomes[indice] + " Foi removido");

        nomes[indice] = string.Empty;
        alturas[indice] = 0;
    }
    else
    {
        Console.WriteLine("A pilha está vazia.");
    }
}
Push("André", 1.71);
Push("Felipe", 1.80);
Push("Thiago", 1.50);
Push("Paulo", 1.65);
Push("José", 1.72);

Console.WriteLine("===================================================");

Console.WriteLine("O topo da pilha é: ");
Console.WriteLine("Nome: " + nomes[indice - 1]);
Console.WriteLine("Altura: " + alturas[indice - 1]);


Pop();

Console.WriteLine("===================================================");

Console.WriteLine("O topo da pilha é: ");
Console.WriteLine("Nome: " + nomes[indice - 1]);
Console.WriteLine("Altura: " + alturas[indice - 1]);
