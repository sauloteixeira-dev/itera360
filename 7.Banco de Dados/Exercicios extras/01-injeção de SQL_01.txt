CREATE DATABASE EscolaDB;
GO

USE EscolaDB;
GO

------------------------------------------------------------
-- 2️⃣ CRIAÇÃO DAS TABELAS
------------------------------------------------------------

CREATE TABLE Alunos (
    IdAluno INT IDENTITY(1,1) PRIMARY KEY,
    NomeAluno VARCHAR(100) NOT NULL,
    Idade INT CHECK (Idade >= 5),
    DataNascimento DATE,
    Cidade VARCHAR(80),
    Estado CHAR(2)
);

CREATE TABLE Professores (
    IdProfessor INT IDENTITY(1,1) PRIMARY KEY,
    NomeProfessor VARCHAR(100) NOT NULL,
    IdadeProfessor INT,
    SalarioProfessor DECIMAL(10,2),
    EspecialidadeProfessor VARCHAR(100)
);

CREATE TABLE Cursos (
    IdCurso INT IDENTITY(1,1) PRIMARY KEY,
    NomeCurso VARCHAR(100) NOT NULL,
    CargaHoraria INT,
    IdProfessor INT,
    FOREIGN KEY (IdProfessor) REFERENCES Professores(IdProfessor)
);

CREATE TABLE Matriculas (
    IdMatricula INT IDENTITY(1,1) PRIMARY KEY,
    IdAluno INT,
    IdCurso INT,
    DataMatricula DATE DEFAULT GETDATE(),
    FOREIGN KEY (IdAluno) REFERENCES Alunos(IdAluno),
    FOREIGN KEY (IdCurso) REFERENCES Cursos(IdCurso)
);

CREATE TABLE Notas (
    IdNota INT IDENTITY(1,1) PRIMARY KEY,
    IdAluno INT,
    IdCurso INT,
    Nota DECIMAL(4,2),
    DataLancamento DATE DEFAULT GETDATE(),
    FOREIGN KEY (IdAluno) REFERENCES Alunos(IdAluno),
    FOREIGN KEY (IdCurso) REFERENCES Cursos(IdCurso)
);
